-- [[ 🛡️ SAJIB TOOLS OFFICIAL - DECODED ]]

local Config = {
    -- ডিকোড করা লিঙ্ক: https://raw.githubusercontent.com/Norisk-Bro/Song-edit/refs/heads/main/expire_date.txt
    [1] = "https://raw.githubusercontent.com/Norisk-Bro/Song-edit/refs/heads/main/expire_date.txt",
    [2] = {
        ["January"]=1, ["February"]=2, ["March"]=3, ["April"]=4, ["May"]=5, ["June"]=6, 
        ["July"]=7, ["August"]=8, ["September"]=9, ["October"]=10, ["November"]=11, ["December"]=12
    }
}

local gg = gg
local os = os

-- HWID জেনারেট করার ফাংশন
local function genHWID()
    local info = gg.getTargetInfo()
    local raw = info.nativeLibraryDir:sub(-5) .. info.packageName:sub(1,4)
    local hex = ""
    for i = 1, #raw do
        local char = raw:sub(i,i):upper()
        if char:match("[%w]") then
            hex = hex .. "X" .. char
        end
    end
    return hex .. "X"
end

local myHWID = genHWID()
local req = gg.makeRequest(Config[1])

-- সার্ভার চেক
if not req or req.code ~= 200 then 
    gg.alert("❌ Server Error") 
    os.exit() 
end

local auth = false
local uName, uDate = "", ""

-- ইউজার অথেন্টিকেশন চেক
for line in req.content:gmatch("[^\r\n]+") do
    local name, id, date = line:match("(.+)%+(.+)%+(.+)")
    if id == myHWID then
        auth = true
        uName = name
        uDate = date
        break
    end
end

-- যদি আইডি রেজিস্টার্ড না থাকে
if not auth then
    gg.copyText(myHWID)
    gg.alert("🚫 NOT REGISTERED!\n\n👤 User: Guest\n🔑 HWID: "..myHWID.."\n\n(ID Copied! Send to Sajib Tools Official)")
    os.exit()
end

-- এক্সপায়ারি ডেট চেক
local d, m, y = uDate:match("(%d+)%-(%a+)%-(%d+)")
local mn = Config[2][m] or 0
local now = os.date("*t")

if now.year > tonumber(y) or (now.year == tonumber(y) and now.month > mn) or (now.year == tonumber(y) and now.month == mn and now.day > tonumber(d)) then
    gg.alert("🚫 EXPIRED!\n👤 User: "..uName.."\n📅 Date: "..uDate)
    os.exit()
end

-- মেমোরি প্যাচ ফাংশন (il2cpp.so লাইব্রেরি প্যাচ করার জন্য)
local patch_function = function(offset, hex_values)
    local lib_name = "il2cpp.so" -- ডিকোড করা লাইব্রেরি নাম
    local ranges = gg.getRangesList(gg.getTargetInfo().nativeLibraryDir .. "/lib" .. lib_name)
    for _, v in ipairs(ranges) do
        -- ELF Header চেক (0x464C457F)
        if gg.getValues({{address = v.start, flags = 4}})[1].value == 1179403647 then
            local bytes = {}
            hex_values:gsub("..", function(x) 
                table.insert(bytes, {address = v.start + offset + #bytes, flags = 1, value = x .. "h"}) 
            end)
            gg.setValues(bytes)
        end
    end
end

local state = { pc = false }

gg.alert("━━━━━━━━━━━━━━━━━━━━\n👑 SAJIB TOOLS OFFICIAL\n━━━━━━━━━━━━━━━━━━━━\n👤 Welcome: "..uName.."\n✅ Status: Authorized\n📅 Expire: "..uDate.."\n━━━━━━━━━━━━━━━━━━━━", "START SCRIPT")

-- মেনু ফাংশন
function _menu()
    local sel = gg.choice({state.pc and "🔛 PC [ ON ]" or "📴 PC [ OFF ]", "❌ EXIT"}, nil, "👑 SAJIB TOOLS\n👤 User: "..uName.."\n📅 Expire: "..uDate)
    
    if sel == 1 then
        gg.toast("⏳ Working...")
        if not state.pc then
            -- PC ON করার জন্য প্যাচ (Hex: 00008052C0035FD6)
            patch_function(2019118045, "00008052C0035FD6")
            patch_function(2019118047, "00008052C0035FD6")
            state.pc = true
        else
            -- PC OFF করার জন্য প্যাচ (Hex: 1F2003D5)
            patch_function(68968552, "1F2003D5")
            patch_function(68968968, "1F2003D5")
            state.pc = false
        end
    elseif sel == 2 then 
        os.exit() 
    end
end

-- মেইন লুপ
while true do 
    if gg.isVisible() then 
        gg.setVisible(false) 
        _menu() 
    end 
end
