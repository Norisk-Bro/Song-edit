function MemoryPatcher(lib, offset, hex)
    local info = gg.getTargetInfo()
    local path = info.nativeLibraryDir
    local ranges = gg.getRangesList(path.."/lib"..lib)

    for _, v in ipairs(ranges) do
        local check = gg.getValues({{address = v.start, flags = gg.TYPE_DWORD}})
        if check[1].value == 0x464C457F then
            local addr = v.start + offset
            local bytes = {}
            hex:gsub("..", function(x)
                table.insert(bytes, {address = addr + #bytes, flags = gg.TYPE_BYTE, value = x .. "h"})
            end)
            gg.setValues(bytes)
        end
    end
end
