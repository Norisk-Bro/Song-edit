-- Fetch Expire Date
local _req = gg.makeRequest(_0xUrl)
if not _req or _req.code ~= 200 then
    gg.alert("❌ Server Error: Expire check failed.")
    os.exit()
end

local _server_date = _req.content:gsub("%s+", "")
local _day, _mon_str, _year = _server_date:match("(%d+)%-(%a+)%-(%d+)")
local _mon_num = _0xMonths[_mon_str] or 0
local _cur = os.date("*t")
local _expired = false

if _cur.year > tonumber(_year) then _expired = true
elseif _cur.year == tonumber(_year) then
    if _cur.month > _mon_num then _expired = true
    elseif _cur.month == _mon_num then
        if _cur.day > tonumber(_day) then _expired = true end
    end
end

if _expired then
    gg.alert("🚫 SCRIPT EXPIRED!\n📅 Date: ".._server_date)
    os.exit()
end

-- Memory Patch Function
function _patch(_off, _hex)
    local _info = gg.getTargetInfo()
    if not _info then return end
    local _r = gg.getRangesList(_info.nativeLibraryDir .. "/lib" .. _0xLib)
    for _, v in ipairs(_r) do
        if gg.getValues({{address = v.start, flags = 4}})[1].value == 1179403647 then
            local _b = {}
            _hex:gsub("..", function(x)
                table.insert(_b, {address = v.start + _off + #_b, flags = 1, value = x .. "h"})
            end)
            gg.setValues(_b)
        end
    end
end
