-- HWID GENERATOR
local function genHWID()
    local info = gg.getTargetInfo()
    local raw = info.nativeLibraryDir:sub(-5) .. info.packageName:sub(1,4)
    local hex = ""
    for i = 1, #raw do
        local char = raw:sub(i,i):upper()
        if char:match("[%w]") then
            hex = hex .. "X" .. char
        end
    end
    return hex .. "X"
end
